const scene = new THREE.Scene();
scene.background = new THREE.Color(0x111111);

const camera = new THREE.PerspectiveCamera(50, window.innerWidth / window.innerHeight, 1, 8000);
camera.position.set(0, 800, 0);
camera.lookAt(0, 0, 0);

const cameraRig = new THREE.Object3D();
cameraRig.lookAt(0,0,0);
cameraRig.add(camera);
scene.add(cameraRig);

const renderer = new THREE.WebGLRenderer({ canvas: document.querySelector('#canvas'), antialias: true });
renderer.setSize(
  renderer.domElement.clientWidth,
  renderer.domElement.clientHeight,
  false
);
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));

const controls = new OrbitControls(cameraRig, renderer.domElement);
controls.enableDamping = true;
controls.dampingFactor = 0.08;
controls.minDistance   = VIEW_CLOSE;
controls.maxDistance   = 3000;
controls.minPolarAngle = 0;
controls.maxPolarAngle = Math.PI / 2;
